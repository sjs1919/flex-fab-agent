# 评估体系语义层修正 + 分层基线 实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 修正评估体系语义层的「无 context 保守给 0」失真——无 context 的 case faithfulness 不计权、answer_relevancy 用真实 judge 独立打分；新增分层基线诊断报告（工具/轨迹/语义各设阈值），综合 7/10 仍是门禁。

**架构：** 三处改动——① `judge.py` 返回 `has_context` 字段，无 context 时 faithfulness 标记未评估（不计权）、relevancy 走真实 judge；② `runner.py` 聚合时按 has_context 自适应加权（有 context 用两指标均值，无 context 只用 relevancy）；③ `print_summary` 增加分层健康度输出，语义层仅统计有 context 的 case。

**技术栈：** Python 3.12 / pytest / 现有 demo 三层评估（`demo/eval/`）。

**依据：** `docs/demo/09-reports/demo-评估改造方案-2026-08-09.md` 第 9.7 节（用户已确认的三项决策：保留权重 / 分层阈值合适 / 分层基线作诊断报告）。

JUDGE_RELEVANCY_SYSTEM_PROMPT = """你是一名专业的问答评估专家。请对以下 Agent 回答进行评分。

评分维度：
answer_relevancy（相关性，0-1）：回答是否直接、完整地回答了用户的问题。
0.0 = 答非所问，1.0 = 精准命中问题核心。

只输出一个 JSON 对象，格式：
{"answer_relevancy": 0.8}

不要输出任何其他内容。
"""


def build_relevancy_messages(question: str, answer: str) -> list[dict]:

{answer}

请打分。"""
    return [
        {"role": "system", "content": JUDGE_RELEVANCY_SYSTEM_PROMPT},
        {"role": "user", "content": user_content},
    ]

```python
from .judge_prompt import build_judge_messages, build_relevancy_messages

def judge_semantic_quality(question, context, answer):
    if not context:
        # 无检索上下文：faithfulness 无法评估（标记未评估，不计权），
        # relevancy 用真实 judge 独立打分（不再用关键词启发式）
        messages = build_relevancy_messages(question, answer)
        try:
            response = call_llm(messages, max_tokens=200, temperature=0.0)
            raw = response.choices[0].message.content or ""
            parsed = parse_judge_response(raw)
            return {
                "faithfulness": 0.0,
                "faithfulness_evaluated": False,
                "answer_relevancy": parsed.get("answer_relevancy", 0.0),
                "has_context": False,
            }
        except Exception as e:
            print(f"  ⚠️  [judge] LLM 打分失败，降级关键词启发式: {type(e).__name__}")
            return {
                "faithfulness": 0.0,
                "faithfulness_evaluated": False,
                "answer_relevancy": _judge_relevancy_only(question, answer),
                "has_context": False,
            }

    messages = build_judge_messages(question, context, answer)
    try:
        response = call_llm(messages, max_tokens=200, temperature=0.0)
        raw = response.choices[0].message.content or ""
        parsed = parse_judge_response(raw)
        return {
            "faithfulness": parsed.get("faithfulness", 0.0),
            "faithfulness_evaluated": True,
            "answer_relevancy": parsed.get("answer_relevancy", 0.0),
            "has_context": True,
        }
    except Exception as e:
        print(f"  ⚠️  [judge] LLM 打分失败，降级 0 分: {type(e).__name__}")
        return {
            "faithfulness": 0.0,
            "faithfulness_evaluated": False,
            "answer_relevancy": 0.0,
            "has_context": True,
        }
```

> **注意：** 保留 `_judge_relevancy_only`（降级兜底仍用它），但正常路径无 context 时也走真实 judge。`parse_judge_response` 需支持只含 `answer_relevancy` 的 JSON（当前返回 dict 已含两键，relevancy-only 时 faithfulness 取默认 0）。

```python
def print_summary(results):
    if not results:
        return {"passed": 0, "total": 0, "avg_score": 0,
                "layer_health": {"tool": None, "trajectory": None, "semantic": None}}

    passed = sum(1 for r in results if r["pass"])
    avg_score = sum(r["overall_score"] for r in results) / len(results)
    avg_tool = sum(r["tool"]["overall_score"] for r in results) / len(results)
    avg_traj = sum(r["trajectory"]["trajectory_score"] for r in results) / len(results)

    # 语义层：仅统计有 context 的 case（无 context 的 case faithfulness 未评估，不参与）
    context_cases = [r for r in results if r["semantic"].get("has_context")]
    if context_cases:
        avg_sem = sum(
            (r["semantic"]["faithfulness"] + r["semantic"]["answer_relevancy"]) / 2
            for r in context_cases
        ) / len(context_cases)
    else:
        avg_sem = None
    n_context = len(context_cases)

    # 各层阈值
    THRESHOLDS = {"tool": 0.7, "trajectory": 0.6, "semantic": 0.5}
    layer_health = {
        "tool": {"avg": round(avg_tool, 2), "threshold": 0.7, "ok": avg_tool >= 0.7, "n": len(results)},
        "trajectory": {"avg": round(avg_traj, 2), "threshold": 0.6, "ok": avg_traj >= 0.6, "n": len(results)},
        "semantic": {"avg": round(avg_sem, 2) if avg_sem is not None else None,
                     "threshold": 0.5, "ok": avg_sem is not None and avg_sem >= 0.5, "n": n_context},
    }

    print(f"\n{'=' * 60}\n 📊 评估汇总\n{'=' * 60}")
    print(f" 通过: {passed}/{len(results)} ({passed / len(results) * 100:.0f}%)")
    print(f" 综合评分: {avg_score:.2f}")
    print(f" 工具层: {avg_tool:.2f} | 轨迹层: {avg_traj:.2f}"
          + (f" | 语义层: {avg_sem:.2f}（{n_context} case 有 context）" if avg_sem is not None else " | 语义层: N/A（0 case 有 context）"))

    # 综合门禁（业务交付指标）
    BASELINE_PASS = 7
    print(f"\n 回归检查：")
    if passed >= BASELINE_PASS:
        print(f"   ✅ 综合通过 {passed}/{len(results)}（overall ≥0.6）")
    else:
        print(f"   ❌ 未通过综合门禁（{passed}/{len(results)} < {BASELINE_PASS}/{len(results)}）")

    # 分层健康度（工程诊断输出，不判 fail）
    print(f"  分层健康度：")
    for name, h in layer_health.items():
        if h["avg"] is None:
            print(f"   ⚠️  {name}层: N/A（0 case 有 context）")
        else:
            mark = "✅" if h["ok"] else "⚠️"
            print(f"   {mark} {name}层 {h['avg']:.2f} {'≥' if h['ok'] else '<'} {h['threshold']}（{h['n']} case 参与）")

    return {
        "passed": passed, "total": len(results), "avg_score": avg_score,
        "avg_tool": avg_tool, "avg_traj": avg_traj, "avg_sem": avg_sem,
        "layer_health": layer_health,
    }
```
